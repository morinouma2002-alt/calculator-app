    function mixWithWhite(hex, amount) {
      const color = hex.replace("#", "");
      const r = parseInt(color.substring(0, 2), 16);
      const g = parseInt(color.substring(2, 4), 16);
      const b = parseInt(color.substring(4, 6), 16);

      const newR = Math.round(r + (255 - r) * amount);
      const newG = Math.round(g + (255 - g) * amount);
      const newB = Math.round(b + (255 - b) * amount);

      return "#" +
        newR.toString(16).padStart(2, "0") +
        newG.toString(16).padStart(2, "0") +
        newB.toString(16).padStart(2, "0");
    }

    function mixWithBlack(hex, amount) {
      const color = hex.replace("#", "");
      const r = parseInt(color.substring(0, 2), 16);
      const g = parseInt(color.substring(2, 4), 16);
      const b = parseInt(color.substring(4, 6), 16);

      const newR = Math.round(r * (1 - amount));
      const newG = Math.round(g * (1 - amount));
      const newB = Math.round(b * (1 - amount));

      return "#" +
        newR.toString(16).padStart(2, "0") +
        newG.toString(16).padStart(2, "0") +
        newB.toString(16).padStart(2, "0");
    }

    function applyAdjustedTheme(themeName) {
      const theme = themes[themeName];

      for (const key in theme) {
        let value = theme[key];

        if (value.startsWith("#")) {
          if (currentStrength < 50) {
            const amount = (50 - currentStrength) / 50;
            value = mixWithWhite(value, amount);
          } else if (currentStrength > 50) {
            const amount = (currentStrength - 50) / 100;
            value = mixWithBlack(value, amount);
          }
        }

        document.documentElement.style.setProperty(key, value);
      }
    }

    function changeTheme(themeName) {
      currentThemeName = themeName;
      applyAdjustedTheme(themeName);
      localStorage.setItem("calcTheme", themeName);
    }

    function updateStrength(value) {
      currentStrength = Number(value);
      strengthValue.textContent = value;
      localStorage.setItem("calcStrength", value);
      applyAdjustedTheme(currentThemeName);
    }

    function toggleThemeMenu() {
      themeMenu.classList.toggle("show");
    }

    function selectTheme(themeName) {
      changeTheme(themeName);
      themeMenu.classList.remove("show");
    }

    function appendValue(value) {
      display.value += value;
    }

    function clearDisplay() {
      display.value = "";
    }

    function calculate() {
      try {
        const expression = display.value;
        const result = eval(expression);

        const text = expression + " = " + result;
        addHistoryToUI(text);
        saveHistory(text);

        display.value = result;
      } catch {
        display.value = "Error";
      }
    }

    function percent() {
      display.value = display.value / 100;
    }

    function addHistoryToUI(text) {
      const item = document.createElement("div");
      item.className = "history-item";
      item.textContent = text;

      item.onclick = () => {
        display.value = text.split(" = ")[0];
      };

      historyDiv.prepend(item);
    }

    function saveHistory(text) {
      let list = JSON.parse(localStorage.getItem("calcHistory")) || [];
      list.unshift(text);
      localStorage.setItem("calcHistory", JSON.stringify(list));
    }

    function clearHistory() {
      localStorage.removeItem("calcHistory");
      historyDiv.innerHTML = "";
    }

    function memoryAdd() {
      memory += Number(display.value) || 0;
    }

    function memorySubtract() {
      memory -= Number(display.value) || 0;
    }

    const quizList = [
      { q: "3 + 5", a: "8" },
      { q: "10 - 4", a: "6" },
      { q: "6 × 7", a: "42" },
      { q: "20 ÷ 5", a: "4" }
    ];

    let currentQuiz = 0;

    function toggleQuiz() {
      const box = document.getElementById("quizBox");
      if (box.style.display === "none") {
        box.style.display = "block";
        showQuiz();
      } else {
        box.style.display = "none";
      }
    }

    function showQuiz() {
      document.getElementById("quizQuestion").textContent =
        quizList[currentQuiz].q + " = ?";
      document.getElementById("quizAnswer").value = "";
      document.getElementById("quizResult").textContent = "";
    }

    function checkQuiz() {
      const input = document.getElementById("quizAnswer").value.trim();
      const correct = quizList[currentQuiz].a;

      if (input == correct) {
        document.getElementById("quizResult").textContent = "正解！";
        currentQuiz = (currentQuiz + 1) % quizList.length;
      } else {
        document.getElementById("quizResult").textContent =
          "ちがう（正解: " + correct + "）";
      }
    }
